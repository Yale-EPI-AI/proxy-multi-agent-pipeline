        df = fetch_world_bank_indicator(wb_code)
        if df is not None and len(df) > 0:
            df = df.rename(columns={"value": "proxy_value"})
            df = df.dropna(subset=["proxy_value"])
            print(f"    Got {len(df)} rows from WB ({wb_code})")
            
            # Merge with target (target is 2022-only)
            # Use most recent available proxy year per country
            target_2022 = target[target["year"] == 2022].copy()
            
            # Get most recent proxy value per country
            df_latest = (
                df.sort_values("year")
                .groupby("iso")
                .last()
                .reset_index()[["iso", "proxy_value", "year"]]
            )
            
            merged_test = target_2022.merge(df_latest, on="iso", how="inner")
            merged_test = merged_test.dropna(subset=["value", "proxy_value"])
            
            print(f"    Merged rows: {len(merged_test)}")
            if len(merged_test) >= 20:
                proxy = df_latest.copy()
                proxy_acquired = True
                proxy_indicator = wb_code

    # Functional form test

    # Partial correlation controlling for log(GDP per capita)

            f"n={corr.n_observations}. "
            f"Partial r (controlling log GPC) = {partial.partial_r:.3f} (p={partial.partial_p:.3f}). "

else:
